cmake_minimum_required(VERSION 3.22)
project(cassiopeia_android LANGUAGES CXX)
if(NOT CASSIOPEIA_NATIVE_LIB)
  message(FATAL_ERROR "Set CASSIOPEIA_NATIVE_LIB to the Rust static library for this Android ABI")
endif()
add_library(cassiopeia_core STATIC IMPORTED)
set_target_properties(cassiopeia_core PROPERTIES IMPORTED_LOCATION "${CASSIOPEIA_NATIVE_LIB}")
add_library(cassiopeia_android SHARED cassiopeia_jni.cpp)
target_compile_features(cassiopeia_android PRIVATE cxx_std_17)
target_include_directories(cassiopeia_android PRIVATE ../../crates/cassiopeia-native/include)
target_link_libraries(cassiopeia_android PRIVATE cassiopeia_core log android dl m)
